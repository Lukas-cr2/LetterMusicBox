    <script>
        class BibliotecaMusical {
            constructor() {
                this.canciones = [];
                this.filtroActual = "todos";
                this.cancionParaEliminar = null;
                this.iniciar();
            }

            iniciar() {
                document.getElementById("formCancion").addEventListener("submit", e => {
                    e.preventDefault();
                    this.agregarCancion();
                });

                document.getElementById("buscador").addEventListener("input", () => {
                    this.mostrarCanciones();
                });

                this.mostrarCanciones();
            }

            agregarCancion() {
                const nombre = document.getElementById("nombre").value.trim();
                const artista = document.getElementById("artista").value.trim();
                const tag = document.getElementById("tag").value;
                const log = document.getElementById("log").value.trim();

                if (!nombre || !artista) {
                    alert("Debes completar nombre y artista.");
                    return;
                }

                const cancion = {
                    id: Date.now(),
                    nombre,
                    artista,
                    tag,
                    log
                };

                this.canciones.push(cancion);
                document.getElementById("formCancion").reset();
                this.mostrarCanciones();
            }

            mostrarCanciones() {
                const contenedor = document.getElementById("listaCanciones");
                const busqueda = document.getElementById("buscador").value.toLowerCase();
                contenedor.innerHTML = "";

                const cancionesFiltradas = this.canciones.filter(cancion => {
                    const coincideTag = this.filtroActual === "todos" || cancion.tag === this.filtroActual;
                    const coincideBusqueda = cancion.nombre.toLowerCase().includes(busqueda) || 
                                             cancion.artista.toLowerCase().includes(busqueda);
                    return coincideTag && coincideBusqueda;
                });

                if (cancionesFiltradas.length === 0) {
                    contenedor.innerHTML = "<p>No hay canciones.</p>";
                    return;
                }

                cancionesFiltradas.forEach(cancion => {
                    this.mostrarCancion(cancion, contenedor);
                });
            }

            mostrarCancion(cancion, contenedor) {
                const elemento = document.createElement("div");

                elemento.innerHTML = `
                    <h3>${this.escaparHTML(cancion.nombre)}</h3>
                    <p>Artista: ${this.escaparHTML(cancion.artista)}</p>
                    <p>Tag: ${this.escaparHTML(cancion.tag)}</p>
                    <p><strong>Log:</strong></p>
                    <p>${cancion.log ? this.escaparHTML(cancion.log) : "La canción todavía no tiene log."}</p>
                    <button class="btn-editar">Editar log</button>
                    <button class="btn-eliminar">Eliminar canción</button>
                    <hr>
                `;

                elemento.querySelector(".btn-editar").addEventListener("click", () => {
                    this.editarLog(cancion.id);
                });

                elemento.querySelector(".btn-eliminar").addEventListener("click", () => {
                    this.eliminarCancion(cancion.id);
                });

                contenedor.appendChild(elemento);
            }

            escaparHTML(str) {
                return str.replace(/[&<>'"]/g, tag => ({
                    '&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&#39;', '"': '&quot;'
                }[tag] || tag));
            }

            editarLog(id) {
                // Lógica para editar log
            }

            eliminarCancion(id) {
                // Lógica para eliminar canción
            }
        }

        // Inicializar la app y definir la función global para los botones de tags
        const biblioteca = new BibliotecaMusical();

        function filtrarTag(tag) {
            biblioteca.filtroActual = tag;
            biblioteca.mostrarCanciones();
        }
